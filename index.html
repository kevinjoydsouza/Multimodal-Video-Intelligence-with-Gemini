Multimodal Video Intelligence with Gemini

Build a spatial object grounding engine, semantic action search, and synchronized interactive video player with token-optimized JSON schemas.

1. Architectural Concepts

Traditional computer vision approaches require multi-stage pipelines: running YOLO for bounding boxes, OpenCV Haar/MTCNN for face tracking, separate action classifiers, and speech-to-text models. With multimodal Large Language Models like Gemini 2.5 Flash, video is ingested natively inside a unified spatial-temporal context window.

Sampling & Token Optimization:
Videos uploaded through the Google GenAI Files API are automatically sampled visually at 1 frame per second (1 FPS) (~258–263 tokens/second) with audio consumed at ~32 tokens/second. A 1-minute video costs only ~17,700 tokens, easily fitting inside Gemini's 1M+ context window.

Spatial Entity Grounding Format

Gemini returns normalized 2D coordinates scaled between 0 and 1000:

[ymin, xmin, ymax, xmax] where top-left is (0,0) and bottom-right is (1000, 1000).

To rescale coordinates to pixel boundaries on any video resolution:

  • pixel_x = (xmin / 1000) * image_width
  • pixel_y = (ymin / 1000) * image_height

2. Quickstart & Minimal Spatial Grounding

Before running full pipelines, verify spatial coordinate prediction on a single test video using this 30-line script:

demo_grounding.py 
import os
import time
from dotenv import load_dotenv
from google import genai
from google.genai import types
from pydantic import BaseModel, Field
from typing import List

load_dotenv()
client = genai.Client(api_key=os.environ["GEMINI_API_KEY"])

    xmin: int = Field(description="Left coordinate [0-1000]")
    ymax: int = Field(description="Bottom coordinate [0-1000]")
    xmax: int = Field(description="Right coordinate [0-1000]")

class DetectedEntity(BaseModel):
    label: str
    action: str
    box: Box2D

class FrameAnalysis(BaseModel):
    scene_summary: str
    entities: List[DetectedEntity]

# Ingest and ground entities
print("[*] Uploading video file...")

# Poll the file state until it is ACTIVE

    time.sleep(5)
    video_file = client.files.get(name=video_file.name)

    raise RuntimeError(f"File processing failed: {video_file.error.message}")

print("[*] File is ACTIVE. Running Gemini model content generation...")
response = client.models.generate_content(

3. Backend Pipeline (video_analytics.py)

This script processes any target video, enforces persistent entity tracking across frames (person_1, face_1), extracts attire color/style, crops face thumbnails with OpenCV, and generates an indexed JSON file alongside your video.

video_analytics.py 
import os
import sys
import time
import json
import cv2
from typing import List, Optional
from pydantic import BaseModel, Field
from dotenv import load_dotenv
from google import genai
from google.genai import types

load_dotenv()

class BoundingBox(BaseModel):
    ymin: int = Field(..., description="Top coordinate normalized 0 to 1000")
    xmin: int = Field(..., description="Left coordinate normalized 0 to 1000")
    ymax: int = Field(..., description="Bottom coordinate normalized 0 to 1000")
    xmax: int = Field(..., description="Right coordinate normalized 0 to 1000")

class DetectedEntity(BaseModel):
    entity_id: str = Field(..., description="Persistent identifier across frames (e.g., person_1, face_1)")
    label: str = Field(..., description="Class name")
    name_or_identity: Optional[str] = Field(None, description="Recognized public figure or role")
    clothing_description: Optional[str] = Field(None, description="Detailed attire, color, and garments")
    visual_attributes: Optional[str] = Field(None, description="Distinctive traits (beard, sunglasses, hair)")
    action: Optional[str] = Field(None, description="Action or posture")
    box_2d: Optional[BoundingBox] = Field(None, description="Bounding box [0-1000]")

class FrameAnalysis(BaseModel):
    timestamp_seconds: float = Field(..., description="Timestamp in seconds")
    timestamp_str: str = Field(..., description="MM:SS")
    environment_or_place: Optional[str] = Field(None, description="Location or landmark")
    scene_summary: str = Field(..., description="Detailed description of visual contents and actions")
    entities: List[DetectedEntity] = Field(default_factory=list)

class VideoAnalyticsReport(BaseModel):
    video_title: str
    total_duration_estimate: str
    prominent_locations: List[str] = Field(default_factory=list)
    named_entities: List[str] = Field(default_factory=list)
    frames: List[FrameAnalysis]

def run_video_analytics(video_path: str, output_json: str = None):
    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        raise ValueError("GEMINI_API_KEY not found in environment!")

    if not os.path.exists(video_path):
        raise FileNotFoundError(f"Video file does not exist: {video_path}")

    if output_json is None:
        base_name, _ = os.path.splitext(video_path)
        output_json = f"{base_name}.json"

    client = genai.Client(api_key=api_key)

    print(f"[*] Uploading video file '{video_path}'...")
    video_file = client.files.upload(file=video_path)

    while video_file.state.name == "PROCESSING":
        time.sleep(4)
        video_file = client.files.get(name=video_file.name)

    if video_file.state.name == "FAILED":
        raise RuntimeError(f"Processing failed: {video_file.error.message}")

    print("[*] Video ingested. Running structured multimodal analysis...")

    prompt = """
    You are an expert video understanding engine. Analyze this video frame-by-frame at 1-second intervals.
    1. Identify recognizable celebrities and public figures in 'name_or_identity'.
    2. Note clothing attire, colors, and clothing items in 'clothing_description' (e.g., 'yellow jersey', 'black shirt').
    3. Keep 'entity_id' consistent across frames (e.g., person_1).
    4. Provide normalized bounding boxes in [0, 1000] coordinates for body and face.
    5. Summarize actions and scene settings per frame.
    """

    response = client.models.generate_content(
        model="gemini-2.5-flash",
        contents=[video_file, prompt],
        config=types.GenerateContentConfig(
            response_mime_type="application/json",
            response_schema=VideoAnalyticsReport,
            temperature=0.1,
        ),
    )

    analytics_data = json.loads(response.text)

    # OpenCV face thumbnail extraction
    os.makedirs("extracted_faces", exist_ok=True)
    cap = cv2.VideoCapture(video_path)

    if cap.isOpened():
        width = cap.get(cv2.CAP_PROP_FRAME_WIDTH)
        height = cap.get(cv2.CAP_PROP_FRAME_HEIGHT)

        for frame_idx, frame_item in enumerate(analytics_data.get("frames", [])):
            t_sec = frame_item.get("timestamp_seconds", 0.0)
            cap.set(cv2.CAP_PROP_POS_MSEC, t_sec * 1000.0)
            ret, frame_bgr = cap.read()
            if not ret or frame_bgr is None:
                continue

            for ent_idx, ent in enumerate(frame_item.get("entities", [])):
                lbl = ent.get("label", "").lower()
                if ("face" in lbl or "person" in lbl) and ent.get("box_2d"):
                    box = ent["box_2d"]
                    ymin = max(0, int((box["ymin"] / 1000.0) * height))
                    xmin = max(0, int((box["xmin"] / 1000.0) * width))
                    ymax = min(int(height), int((box["ymax"] / 1000.0) * height))
                    xmax = min(int(width), int((box["xmax"] / 1000.0) * width))

                    if ymax > ymin and xmax > xmin:
                        cropped = frame_bgr[ymin:ymax, xmin:xmax]
                        thumb_path = f"extracted_faces/crop_{frame_idx}_{ent_idx}_{ent.get('entity_id', 'ent')}.jpg"
                        cv2.imwrite(thumb_path, cropped)
                        ent["thumbnail_path"] = thumb_path
        cap.release()

    with open(output_json, "w", encoding="utf-8") as f:
        json.dump(analytics_data, f, indent=2)

    print(f"[✓] Analysis complete! Saved to: {output_json}")

if __name__ == "__main__":
    if len(sys.argv) > 1:
        run_video_analytics(sys.argv[1])

4. Front-End Interface (app.py)

The front-end integrates a compact 3-column video gallery, multi-word semantic fuzzy search, and an HTML5 player with a YouTube-style scrubber showing in-timeline glowing green match segments and live-updating canvas overlays.

app.py 
import streamlit as st
import json
import os
import re
import cv2
import base64
import streamlit.components.v1 as components
from PIL import Image

st.set_page_config(page_title="Multimodal Video Intelligence", page_icon="🎬", layout="wide")
st.title("🎬 Multimodal Video Intelligence Studio")

with st.sidebar:
    st.header("📂 Media Directory")
    default_dir = os.path.join(os.getcwd(), "media") if os.path.exists(os.path.join(os.getcwd(), "media")) else os.getcwd()
    media_folder = st.text_input("Folder Path", value=default_dir)

    st.header("🎨 Visual Overlays")
    show_boxes = st.toggle("Enable Bounding Boxes", value=True)
    show_labels = st.toggle("Show Entity Labels", value=True)
    show_attire = st.toggle("Show Clothing / Attire", value=True)
    box_color = st.color_picker("Box & Label Color", "#00FF7F")

STOPWORDS = {"with", "in", "a", "an", "the", "and", "or", "on", "at", "is", "wearing", "to", "of", "by", "for"}
SYNONYM_GROUPS = [
    {"tshirt", "t shirt", "shirt", "jersey", "uniform", "clothing", "dress", "outfit"},
    {"speaking", "talking", "speech", "saying", "mic"},
    {"driving", "car", "vehicle", "ferrari", "auto"},
    {"walking", "moving", "running"},
]

def get_synonyms(word: str) -> set:
    results = {word}
    for group in SYNONYM_GROUPS:
        if word in group:
            results.update(group)
    return results

def normalize_text(text: str) -> str:
    cleaned = text.lower().replace("-", " ")
    return re.sub(r"[^\w\s]", " ", cleaned)

@st.cache_data(show_spinner=False)
def load_all_video_catalogs(folder_path: str):
    catalog = []
    if not os.path.isdir(folder_path):
        return catalog
    for file_name in sorted(os.listdir(folder_path)):
        if file_name.lower().endswith((".mp4", ".mov", ".webm")):
            base_name, _ = os.path.splitext(file_name)
            json_path = os.path.join(folder_path, f"{base_name}.json")
            video_path = os.path.join(folder_path, file_name)
            if os.path.exists(json_path):
                try:
                    with open(json_path, "r", encoding="utf-8") as jf:
                        catalog.append({"video_name": file_name, "video_path": video_path, "data": json.load(jf)})
                except Exception:
                    pass
    return catalog

def extract_video_frame(video_path: str, timestamp_sec: float, max_width: int = 450):
    if not os.path.exists(video_path):
        return None
    cap = cv2.VideoCapture(video_path)
    cap.set(cv2.CAP_PROP_POS_MSEC, max(0.0, timestamp_sec * 1000.0))
    ret, frame_bgr = cap.read()
    cap.release()
    if ret and frame_bgr is not None:
        h, w = frame_bgr.shape[:2]
        if w > max_width:
            frame_bgr = cv2.resize(frame_bgr, (max_width, int(h * (max_width / float(w)))), interpolation=cv2.INTER_AREA)
        return Image.fromarray(cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB))
    return None

def get_video_base64(video_path: str) -> str:
    with open(video_path, "rb") as f:
        return base64.b64encode(f.read()).decode("utf-8")

video_catalog = load_all_video_catalogs(media_folder)

if not video_catalog:
    st.warning(f"No indexed videos found in `{media_folder}`. Ensure both .mp4 and matching .json exist.")
    st.stop()

st.subheader("🔍 Search Video Repository")
search_query = st.text_input("Search by person, action, clothing, or scene keyword:", value="").strip()

processed_videos = []
norm_query = normalize_text(search_query) if search_query else ""
query_tokens = [t for t in norm_query.split() if t not in STOPWORDS] or norm_query.split()

for video_item in video_catalog:
    v_name = video_item["video_name"]
    frames = video_item["data"].get("frames", [])
    matching_indices = []

    if search_query:
        for f_idx, fr in enumerate(frames):
            text_parts = [v_name, fr.get("scene_summary", ""), fr.get("environment_or_place", "")]
            for ent in fr.get("entities", []):
                text_parts.extend([ent.get("label", ""), ent.get("name_or_identity", ""), ent.get("clothing_description", ""), ent.get("action", "")])
            frame_corpus = normalize_text(" ".join(filter(None, text_parts)))
            score = sum(1 for token in query_tokens if any(syn in frame_corpus for syn in get_synonyms(token)))
            if score > 0:
                matching_indices.append(f_idx)

        if matching_indices:
            processed_videos.append({"video_name": v_name, "video_path": video_item["video_path"], "matching_indices": matching_indices, "first_match_idx": matching_indices[0], "video_data": video_item["data"]})
    else:
        default_idx = min(max(1, len(frames) // 3), max(0, len(frames) - 1))
        processed_videos.append({"video_name": v_name, "video_path": video_item["video_path"], "matching_indices": [], "first_match_idx": default_idx, "video_data": video_item["data"]})

if "active_video_name" not in st.session_state or st.session_state["active_video_name"] not in [v["video_name"] for v in processed_videos]:
    st.session_state["active_video_name"] = processed_videos[0]["video_name"] if processed_videos else None
    st.session_state["active_initial_sec"] = processed_videos[0]["first_match_idx"] if processed_videos else 0

st.markdown(f"### 🎞️ Video Gallery ({len(processed_videos)})")
if not processed_videos:
    st.info("No videos matched the query.")
else:
    cols = st.columns(3)
    for idx, item in enumerate(processed_videos):
        v_name = item["video_name"]
        frames = item["video_data"].get("frames", [])
        preview_sec = frames[item["first_match_idx"]].get("timestamp_seconds", 1.0) if frames else 0.0
        is_selected = (v_name == st.session_state.get("active_video_name"))

        with cols[idx % 3]:
            with st.container(border=True):
                thumb = extract_video_frame(item["video_path"], preview_sec, max_width=400)
                if thumb:
                    st.image(thumb, use_container_width=True)
                st.markdown(f"**🎥 {v_name}**")
                if item["matching_indices"]:
                    st.caption(f"🎯 **{len(item['matching_indices'])} matching scene(s)**")
                if st.button("✅ Selected" if is_selected else "▶️ Play in Player", key=f"sel_{idx}_{v_name}", disabled=is_selected):
                    st.session_state["active_video_name"] = v_name
                    st.session_state["active_initial_sec"] = item["first_match_idx"]
                    st.rerun()

st.divider()

active_entry = next((v for v in processed_videos if v["video_name"] == st.session_state.get("active_video_name")), None)
if active_entry:
    st.subheader(f"🎬 Now Playing: {active_entry['video_name']}")
    video_b64 = get_video_base64(active_entry["video_path"])
    frames_json = json.dumps(active_entry["video_data"].get("frames", []))
    matching_json = json.dumps(active_entry["matching_indices"])

    player_html = f"""
    
    
    
        
        
    
    
        

            

                
                
            

            

                

                    

                    

                

                

                    

                        
                        00:00 / 00:00
                    

                    
Ready...

                

            

        

        
    
    
    """
    components.html(player_html, height=660)

Bash
python3 -m venv .venv
source .venv/bin/activate

Terminal

streamlit run app.py
